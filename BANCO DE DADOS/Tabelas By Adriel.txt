CREATE TABLE plano_assinatura(
id_plano SERIAL NOT NULL,

precosemestral_plano numeric(10, 2),

CREATE TABLE pedido (

-- Função que atualiza o estoque do produto
CREATE OR REPLACE FUNCTION atualizar_estoque_produto()
RETURNS TRIGGER AS $$
BEGIN
  -- Atualiza o estoque do produto baseado na quantidade do item pedido
  UPDATE produto
  SET estoque_produto = estoque_produto - NEW.quantidade_item
  WHERE id_produto = NEW.id_produto;
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger que chama a função após a inserção de um item_pedido
CREATE TRIGGER trigger_atualizar_estoque_produto
AFTER INSERT ON item_pedido
FOR EACH ROW
EXECUTE FUNCTION atualizar_estoque_produto();
