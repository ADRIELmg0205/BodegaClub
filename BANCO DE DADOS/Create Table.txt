CREATE TABLE assinatura (
id_assinatura SERIAL NOT NULL, 
id_plano int4 NOT NULL, 
id_usuario int4 NOT NULL, 
id_vendedor int4 NOT NULL, 
data_assinatura date, 
PRIMARY KEY (id_assinatura)
);

CREATE TABLE carrinho (
id_carrinho SERIAL NOT NULL, 
id_usuario int4 NOT NULL, 
id_produto int4 NOT NULL, 
quantidade_carrinho int4, 
CONSTRAINT quantidade_carrinho 
PRIMARY KEY (id_carrinho)
);

CREATE TABLE endereco (
id_endereco SERIAL NOT NULL, 
cep_endereco numeric(8, 0), 
pais_endereco varchar(100), 
estado_endereco varchar(100), 
cidade_endereco varchar(100), 
bairro_endereco varchar(100) NOT NULL, 
numero_endereco int4 NOT NULL, 
referencia_endereco varchar(255) NOT NULL, 
id_usuario int4 NOT NULL, 
CONSTRAINT id_endereco PRIMARY KEY (id_endereco)
);

CREATE TABLE favoritos (
id_favoritos SERIAL NOT NULL, 
id_produto int4 NOT NULL, 
id_usuario int4 NOT NULL, 
PRIMARY KEY (id_favoritos)
);

CREATE TABLE item_assinatura (
id_itemassinatura SERIAL NOT NULL, 
quantidade_itemassinatura int4, 
id_produto int4 NOT NULL, 
id_assinatura int4 NOT NULL, 
data_itemassinatura date NOT NULL, 
CONSTRAINT q PRIMARY KEY (id_itemassinatura)
);

CREATE TABLE item_pedido (
id_itempedido SERIAL NOT NULL, 
quantidade_itempedido int4, 
id_pedido int4 NOT NULL, 
id_produto int4 NOT NULL, 
PRIMARY KEY (id_itempedido)
);

CREATE TABLE pedido (
id_pedido SERIAL NOT NULL, 
observacao_pedido varchar(255) NOT NULL, 
id_usuario int4 NOT NULL, 
id_vendedor int4 NOT NULL, 
data_pedido date, 
PRIMARY KEY (id_pedido)
);

CREATE TABLE plano_assinatura (
id_plano SERIAL NOT NULL, 
descricao_plano varchar(255), 
preco_plano numeric(10, 2), 
PRIMARY KEY (id_plano)
);

CREATE TABLE produto (
id_produto SERIAL NOT NULL, 
nome_produto varchar(255), 
preco_produto numeric(10, 2), 
descricao_produto varchar(255) NOT NULL, 
tipo_produto varchar(50) NOT NULL, 
imagem_produto varchar(255) NOT NULL, 
principal_produto char(1) NOT NULL, 
estoque_produto int4 NOT NULL, 
CONSTRAINT tipo_produto PRIMARY KEY (id_produto)
);

CREATE TABLE usuario (
id_usuario SERIAL NOT NULL, 
nome_usuario varchar(255), 
email_usuario varchar(255), 
PRIMARY KEY (id_usuario)
);

CREATE TABLE vendedor (
id_vendedor SERIAL NOT NULL, 
id_usuario int4 NOT NULL, 
PRIMARY KEY (id_vendedor)
);
